     import os
     from typing import List, Dict, Optional
     from datetime import datetime, timezone
     from sqlalchemy import select
     from langchain_openai import OpenAIEmbeddings
     from utils.log_util import get_request_id, logger
     from x_models.id_x_010.src.models.x_models import (

     )

     _embeddings_cache = {}

     def get_embeddings():
         """复用 external_provider 表配置,缓存 embeddings 客户端。"""
         cfg = get_external_model_config()
         base_url = cfg.get("embd_base_url") or cfg.get("base_url")
         api_key = cfg.get("api_key")
         model = os.getenv("EXTERNAL_EMBEDDING_MODEL", "Qwen/Qwen3-Embedding-8B")
         key = f"{base_url}|{api_key}|{model}"
         if key not in _embeddings_cache:
             _embeddings_cache[key] = OpenAIEmbeddings(
                 model=model, openai_api_key=api_key, openai_api_base=base_url,
             )
         return _embeddings_cache[key]

     async def embed(text: str) -> List[float]:
         """文本→向量,失败抛错由调用方降级。"""
         emb = get_embeddings()
         return await emb.aembed_query(text)

     async def upsert_table_comment(
         db_name: str, schema_name: str, table_name: str,
         comment: str, columns_desc: dict
     ) -> None:
         """写入/更新一行 table_comments,自动生成 embedding。"""
         text_for_emb = f"{table_name} {comment or ''} {' '.join(columns_desc.values())}"
         vec = await embed(text_for_emb)
         async with get_session()() as session:
             existing = await session.execute(
                 select(TableComment).where(
                     TableComment.db_name == db_name,
                     TableComment.table_name == table_name,
                 )
             )
             row = existing.scalar_one_or_none()
             if row is None:
                 row = TableComment(db_name=db_name, schema_name=schema_name, table_name=table_name)
                 session.add(row)
             row.comment = comment
             row.columns_desc = columns_desc
             row.embedding = vec
             row.last_indexed_at = datetime.now(timezone.utc)
             await session.commit()

     async def search_tables(query: str, db_name: str, top_k: int = 5) -> List[Dict]:
         """从 table_comments 召回相关表。embedding 失败时降级返回全表前 N。"""
         try:
             qvec = await embed(query)
         except Exception as e:
             logger.warning(f"{get_request_id()}embed 失败,降级全表返回: {e}")
             async with get_session()() as session:
                 rows = await session.execute(
                     select(TableComment.table_name, TableComment.comment)
                     .where(TableComment.db_name == db_name).limit(top_k * 2)
                 )
                 return [{"table_name": r[0], "comment": r[1], "score": 0.0} for r in rows.all()]
         async with get_session()() as session:
             try:
                 stmt = (
                     select(TableComment, TableComment.embedding.cosine_distance(qvec).label("dist"))
                     .where(TableComment.db_name == db_name)
                     .order_by("dist").limit(top_k)
                 )
                 rows = await session.execute(stmt)
                 return [{"table_name": r[0].table_name, "comment": r[0].comment, "score": 1 - r[1]} for r in rows.all()]
             except Exception:
                 rows = await session.execute(
                     select(TableComment).where(TableComment.db_name == db_name)
                 )
                 return [{"table_name": r.table_name, "comment": r.comment, "score": 0.0} for r in rows.scalars().all()[:top_k]]

     """数据库 schema 索引构建器:遍历 information_schema,为每张表生成中文描述并写入 table_comments。"""
     from typing import Optional
     from sqlalchemy import text
     from utils.log_util import get_request_id, logger
     from x_models.id_x_010.src.utils.x_vectorstore import upsert_table_comment

     SCHEMA_QUERY = """
     SELECT
         c.table_name,
         pgd.description AS table_comment,
         c.column_name,
         pg_catalog.col_description(c.table_name::regclass, c.ordinal_position) AS col_comment,
         c.data_type
     FROM information_schema.columns c
     LEFT JOIN pg_catalog.pg_statio_all_tables st
         ON st.relname = c.table_name AND st.schemaname = c.table_schema
     LEFT JOIN pg_catalog.pg_description pgd
         ON pgd.objoid = st.relid AND pgd.objsubid = 0
     WHERE c.table_schema = 'public'
     ORDER BY c.table_name, c.ordinal_position;
     """

     async def index_database(db_name: str, db_conn) -> int:
         """遍历目标库 information_schema,为每张表生成描述并 upsert 到 table_comments。
         db_conn: 已连接的 DatabaseConnection(SQL Agent 的)。返回索引表数。"""
         rows, _ = await db_conn.execute_query(SCHEMA_QUERY, internal=True)
         grouped: dict = {}
         for r in rows:
             tname = r["table_name"]
             grouped.setdefault(tname, {"comment": r.get("table_comment") or "", "cols": {}})
             col_desc = r.get("col_comment") or r["column_name"]
             grouped[tname]["cols"][r["column_name"]] = f"{col_desc} ({r['data_type']})"

         from x_models.id_x_010.src.agents.x_sql_agent import PROTECTED_TABLES
         count = 0
         for tname, info in grouped.items():
             if tname.lower() in PROTECTED_TABLES:
                 continue
             comment = info["comment"] or tname
             try:
                 await upsert_table_comment(db_name, "public", tname, comment, info["cols"])
                 count += 1
             except Exception as e:
                 logger.warning(f"{get_request_id()}index_table {tname} 失败: {e}")
         logger.info(f"{get_request_id()}index_database db={db_name} indexed={count}/{len(grouped)}")
         return count
     ```

     """进程内滑动窗口限流器。
     注意:多副本部署时各副本独立计数,无法跨进程聚合。如需精确限流请后续引入 Redis。"""
     import time
     from collections import defaultdict
     from typing import Optional
     from utils.log_util import get_request_id, logger

     RATE_LIMITS = {"chat": 20, "stream_chat": 20, "login": 5}
     _buckets: dict = defaultdict(list)  # key → [timestamp1, timestamp2, ...]
     WINDOW_SEC = 60

     async def rate_limit(scope: str, key: str, limit: Optional[int] = None) -> bool:
         """返回 True 表示允许,False 表示已达上限。"""
         limit = limit or RATE_LIMITS.get(scope, 20)
         now = time.time()
         k = f"{scope}:{key}"
         # 清理窗口外
         _buckets[k] = [t for t in _buckets[k] if now - t < WINDOW_SEC]
         if len(_buckets[k]) >= limit:
             logger.warning(f"{get_request_id()}rate_limit 拒绝: scope={scope} key={key} count={len(_buckets[k])}")
             return False
         _buckets[k].append(now)
         return True

     def clear_rate_limit(scope: str, key: str) -> None:
         """登录成功后清除该 key 的失败计数。"""
         _buckets.pop(f"{scope}:{key}", None)
     ```

     from x_models.id_x_010.src.agents.x_llm_error import friendly_llm_error
     from x_models.id_x_010.src.models.x_models import get_external_model_config

     class VLAgent:
         def __init__(self, model_name: Optional[str] = None, api_key: Optional[str] = None, base_url: Optional[str] = None):
             cfg = get_external_model_config()

         async def stream_analyze(self, image_b64: str, query: str) -> AsyncGenerator[Dict, None]:

             except Exception as e:

  3. `[IN] ops/init.sql]` users 表创建后追加:
     ```sql
     ALTER TABLE users ADD COLUMN IF NOT EXISTS password_algo VARCHAR(16) DEFAULT 'bcrypt';
     COMMENT ON COLUMN users.password_algo IS '密码哈希算法: bcrypt / sha256(老数据)';
     ```
  4. `[IN] src/views/x_views.py]`:
     ```python
     import bcrypt
     import hashlib

     def hash_password(p: str) -> str:
         return bcrypt.hashpw(p.encode(), bcrypt.gensalt(rounds=12)).decode()

     def verify_password(p: str, hash_str: str, algo: str = "bcrypt") -> bool:
         if algo == "bcrypt":
             try: return bcrypt.checkpw(p.encode(), hash_str.encode())
             except ValueError: return False
         elif algo == "sha256":
             return hashlib.sha256(p.encode()).hexdigest() == hash_str
         return False
     ```
  5. `[IN] src/controllers/x_controllers.py]` `login` 校验密码后:
     ```python
     if not verify_password(input.password, user.password_hash, user.password_algo or "sha256"):
         return AuthResponseType(success=False, message="密码错误")
     # 老 sha256 自动升级
     if (user.password_algo or "sha256") != "bcrypt":
         user.password_hash = hash_password(input.password)
         user.password_algo = "bcrypt"
         await session.commit()
     ```
